**MyADC.c**
```c
#include "MyADC.h"

uint16_t ADC_DMA_Buffer[ADC_DMA_BUFFER_SIZE];

void MyADC_Init(void) {
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA | RCC_APB2Periph_ADC1, ENABLE);
	RCC_AHBPeriphClockCmd(RCC_AHBPeriph_DMA1, ENABLE);  //开启DMA时钟
	
	
	GPIO_InitTypeDef GPIO_InitStructure;
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_0;
	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AIN;
	GPIO_Init(GPIOA, &GPIO_InitStructure);
	
	RCC_ADCCLKConfig(RCC_PCLK2_Div6);  //将ADC时钟分频
	
	//初始化DMA1外设
	DMA_InitTypeDef DMA_InitStructure;
	DMA_DeInit(DMA1_Channel1);
	DMA_InitStructure.DMA_PeripheralBaseAddr = (uint32_t)&ADC1->DR;
	DMA_InitStructure.DMA_MemoryBaseAddr = (uint32_t)ADC_DMA_Buffer;
	DMA_InitStructure.DMA_DIR = DMA_DIR_PeripheralSRC;  //?
	DMA_InitStructure.DMA_BufferSize = ADC_DMA_BUFFER_SIZE;
	DMA_InitStructure.DMA_PeripheralInc = DMA_PeripheralInc_Disable;
	DMA_InitStructure.DMA_MemoryInc = DMA_MemoryInc_Enable;
	DMA_InitStructure.DMA_PeripheralDataSize = DMA_PeripheralDataSize_HalfWord;
	DMA_InitStructure.DMA_MemoryDataSize = DMA_MemoryDataSize_HalfWord;
	DMA_InitStructure.DMA_Mode = DMA_Mode_Circular;
	DMA_InitStructure.DMA_Priority = DMA_Priority_High;
	DMA_InitStructure.DMA_M2M = DMA_M2M_Disable;
	DMA_Init(DMA1_Channel1, &DMA_InitStructure);
	
	DMA_Cmd(DMA1_Channel1, ENABLE);
	
	ADC_InitTypeDef ADC_InitStructure;
	ADC_InitStructure.ADC_Mode = ADC_Mode_Independent;
	ADC_InitStructure.ADC_ScanConvMode = DISABLE;
	ADC_InitStructure.ADC_ContinuousConvMode = ENABLE;
	ADC_InitStructure.ADC_ExternalTrigConv = ADC_ExternalTrigConv_None;
	ADC_InitStructure.ADC_DataAlign = ADC_DataAlign_Right;
	ADC_InitStructure.ADC_NbrOfChannel = 1;
	ADC_Init(ADC1, &ADC_InitStructure);
	
	ADC_DMACmd(ADC1, ENABLE);
	
	ADC_Cmd(ADC1, ENABLE);	
	
	ADC_ResetCalibration(ADC1);
	while (ADC_GetResetCalibrationStatus(ADC1));
	
	ADC_StartCalibration(ADC1);
	while (ADC_GetCalibrationStatus(ADC1));
	
	ADC_SoftwareStartConvCmd(ADC1, ENABLE);
}

uint16_t MyADC_GetValue(void) {
	
	while (ADC_GetFlagStatus(ADC1, ADC_FLAG_EOC) == RESET);
	
	return ADC_GetConversionValue(ADC1);
	
}

uint16_t MyADC_GetAverageValue(uint8_t Samples){
	uint32_t Sum = 0;
	for (uint8_t i = 0; i < Samples; ++i ) {
		Sum += MyADC_GetValue();
	}
	return Sum / Samples;
}

float ADC_GetVoltage(void) {
	uint16_t ADC_Value;
	
	ADC_Value = MyADC_GetValue();
	
	return (float)ADC_Value * 3.3f / 4095.0f;
	
}
```

**1. extern**

这是一个关键字，目的是将一个变量复用，就比如在main.c和MyADC.c里面如果定义了两个tmp变量，那么他们是独立的，但是通过extern修饰之后，就告诉编译器，这个变量已经定义过了，在这里直接拿来用不需要重新定义
好处：避免重复定义全局变量，让所有文件都可以共享这一个变量，让工程更加模块化

==注意==
不要漏了
```c
ADC_DMACmd(ADC1, ENABLE);
```

**2. DMA_InitStructure参数**

1. DMA_PeripheralBaseAddr
获取数据的地址，我们需要从这里读取数据，也就是数据从何而来
2. DMA_MemoryBaseAddr
获取到的数据存放地址，将获取到的数据存放在这里，也就是数据该放到何处
3. DMA_DIR
数据移动方向，DMA_DIR_PeripheralSRC代表当前外设作为源，也就是从外设搬运数据出去，DMA_DIR_PeripheralDST代表外设作为目的地，将数据搬入外设
4. DMA_BufferSize
数据缓冲区大小
5. DMA_PeripheralInc
外设地址是否需要递增，这里不需要，因为只用到了一个ADC1，地址固定即可
6. DMA_MemoryInc
内存是否需要递增，这里是需要的，因为缓冲区大小是16，需要自动递增将数据存放在不同的位置
7. DMA_PeripheralDataSize，DMA_MemoryDataSize
数据大小，Byte是8位，Word是32位，HalfWord是16位，ADC1通道是12位的，所以需要HalfWord
8. DMA_Mode
DMA的模式，现在设置了循环模式，当缓冲区数组到头之后，自动回到头
9. DMA_Priority
优先级
10. DMA_M2M
是否是内存->内存，这个项目里面是外设->内存，所以不需要打开
