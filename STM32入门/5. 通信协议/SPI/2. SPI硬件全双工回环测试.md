```c
    volatile uint32_t i;

void LED_Init(void) {
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC, ENABLE);
	
	GPIO_InitTypeDef GPIO_InitStructure;
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_13;
	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_PP;
	GPIO_InitStructure.GPIO_Speed = GPIO_Speed_2MHz;
	GPIO_Init(GPIOC, &GPIO_InitStructure);
	GPIO_SetBits(GPIOC, GPIO_Pin_13);

	while (1) {
		Debug_SendData1 = 0xAA;
```

```c
#ifndef __MYSPI_H

#include "stm32f10x.h"                  // Device header

typedef enum{
	MySPI_OK = 0,
	
	MySPI_ERROR_TXE_TIMEOUT,
	MySPI_ERROR_RXNE_TIMEOUT,
```

**MySPI.c**
```c
#include "MySPI.h"

#define MySPI_TIMEOUT 100000UL

#define MySPI_CS_LOW() 	GPIO_ResetBits(GPIOA, GPIO_Pin_4)
#define MySPI_CS_HIGH()   GPIO_SetBits(GPIOA, GPIO_Pin_4)

void MySPI_Init(void) {
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA | RCC_APB2Periph_SPI1, ENABLE);
	
	GPIO_InitTypeDef GPIO_InitStructure;
	SPI_InitTypeDef SPI_InitStructure;
	
	//初始化GPIOA
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_4;
	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_PP;
	GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
	GPIO_Init(GPIOA, &GPIO_InitStructure);
	
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_5 | GPIO_Pin_7;
	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;
	GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
	GPIO_Init(GPIOA, &GPIO_InitStructure);
	
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_6;
	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IN_FLOATING;
	GPIO_Init(GPIOA, &GPIO_InitStructure);
	
	MySPI_CS_HIGH();
	SPI_I2S_DeInit(SPI1);
	
	//初始化SPI
	SPI_InitStructure.SPI_Direction = SPI_Direction_2Lines_FullDuplex;
	SPI_InitStructure.SPI_Mode = SPI_Mode_Master;
	SPI_InitStructure.SPI_DataSize = SPI_DataSize_8b;
	SPI_InitStructure.SPI_CPOL = SPI_CPOL_Low;
	SPI_InitStructure.SPI_CPHA = SPI_CPHA_1Edge;
	SPI_InitStructure.SPI_FirstBit = SPI_FirstBit_MSB;
	SPI_InitStructure.SPI_BaudRatePrescaler = SPI_BaudRatePrescaler_256;
	SPI_InitStructure.SPI_NSS = SPI_NSS_Soft;
	SPI_InitStructure.SPI_CRCPolynomial = 7;
	SPI_Init(SPI1, &SPI_InitStructure);
	
	SPI_NSSInternalSoftwareConfig(SPI1, SPI_NSSInternalSoft_Set);
	
	SPI_Cmd(SPI1, ENABLE);
}

void MySPI_Start(void) {
	MySPI_CS_LOW();
}

MySPI_Result MySPI_Stop(void) {
	uint32_t Timeout = MySPI_TIMEOUT;
	
	while (SPI_I2S_GetFlagStatus(SPI1, SPI_I2S_FLAG_TXE) == RESET) {
		if (--Timeout == 0) {
			MySPI_CS_HIGH();
			return MySPI_ERROR_TXE_TIMEOUT;
		}
	}
	
	Timeout = MySPI_TIMEOUT;
	while (SPI_I2S_GetFlagStatus(SPI1, SPI_I2S_FLAG_BSY) == SET) {
		if (--Timeout == 0) {
			MySPI_CS_HIGH();
			return MySPI_ERROR_BSY_TIMEOUT;
		}
	}
	
	MySPI_CS_HIGH();
	return MySPI_OK;
}

MySPI_Result MySPI_SwapByte(uint8_t SendData, uint8_t* ReceiveData) {
	uint32_t Timeout = MySPI_TIMEOUT;
	while (SPI_I2S_GetFlagStatus(SPI1, SPI_I2S_FLAG_TXE) == RESET) {
		if (--Timeout == 0) {
			//MySPI_CS_HIGH();
			return MySPI_ERROR_TXE_TIMEOUT;
		}
	}
	SPI_I2S_SendData(SPI1, SendData);
	
	Timeout = MySPI_TIMEOUT;
	while (SPI_I2S_GetFlagStatus(SPI1, SPI_I2S_FLAG_RXNE) == RESET) {  //等待输入缓冲区的数据都进入数据寄存器
		if (--Timeout == 0) {
			//MySPI_CS_HIGH();
			return MySPI_ERROR_RXNE_TIMEOUT;
		}
	}
	*ReceiveData = (uint8_t)SPI_I2S_ReceiveData(SPI1);
	return MySPI_OK;
}
```
