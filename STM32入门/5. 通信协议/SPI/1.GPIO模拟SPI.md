该实验没有真正意义使用SPI外设，只是使用了GPIO口模拟SPI的过程，观察其波形

```c
#include "stm32f10x.h"                  // Device header

#define MySPI_NSS_LOW() GPIO_ResetBits(GPIOA, GPIO_Pin_4)
#define MySPI_NSS_HIGH() GPIO_SetBits(GPIOA, GPIO_Pin_4)
#define MySPI_SCK_LOW() GPIO_ResetBits(GPIOA, GPIO_Pin_5)
#define MySPI_SCK_HIGH() GPIO_SetBits(GPIOA, GPIO_Pin_5)
#define MySPI_MOSI_LOW() GPIO_ResetBits(GPIOA, GPIO_Pin_7)
#define MySPI_MOSI_HIGH() GPIO_SetBits(GPIOA, GPIO_Pin_7)

void MySPI_Delay(void) {
	for (int i = 0; i < 200; ++i); 
}

void MySPI_Init(void) {
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);
	
	GPIO_InitTypeDef GPIO_InitStrucure;
	GPIO_InitStrucure.GPIO_Pin = GPIO_Pin_4 | GPIO_Pin_5 | GPIO_Pin_7;
	GPIO_InitStrucure.GPIO_Mode = GPIO_Mode_Out_PP;
	GPIO_InitStrucure.GPIO_Speed = GPIO_Speed_50MHz;
	GPIO_Init(GPIOA, &GPIO_InitStrucure);
	
	MySPI_NSS_LOW();
	MySPI_SCK_LOW();
	MySPI_MOSI_HIGH();
	
}

void MySPI_SendByte(uint8_t Data) {
	for (int i = 0; i < 8; ++i) {
		if ((Data & 0x80) != 0) {
			MySPI_MOSI_HIGH();
		}else {
			MySPI_MOSI_LOW();
		}
		MySPI_Delay();
		MySPI_SCK_HIGH();
		MySPI_Delay();
		MySPI_SCK_LOW();
		MySPI_Delay();
		Data <<= 1;
	}
}

static void Delay_Long(void){
    volatile uint32_t i;

    for (i = 0; i < 500000; i++);
}	

int main(void) {
	
	MySPI_Init();
		
	while (1) {
		MySPI_NSS_LOW();
		MySPI_SendByte(0xAA);
		MySPI_NSS_HIGH();
		Delay_Long();
		
		MySPI_NSS_LOW();
		MySPI_SendByte(0x55);
		MySPI_NSS_HIGH();
		Delay_Long();
	}
	
}
```
