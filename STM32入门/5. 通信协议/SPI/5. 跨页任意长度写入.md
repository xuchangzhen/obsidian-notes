```c
     * PC13 高电平：LED 熄灭。

	Debug_ProgramResult = W25Q64_WriteDataNoErase(
		TEST_ADDRESS,
		TestData,
		TEST_LENGTH
	);

     * 第五步：
     * 读回刚才写入的数据。

    if (Debug_ProgramResult == W25Q64_OK) {
        Debug_ReadResult = W25Q64_ReadData(
            TEST_ADDRESS,
            Debug_ReadData,
            TEST_LENGTH

    if (Debug_ReadResult == W25Q64_OK) {

                Debug_ProgramCheckPassed = 0;
                Debug_MismatchIndex = (uint8_t)i;
                break;
            }
        }
    }

    /*
     * 所有阶段都正确，点亮 PC13。
     */
    if ((Debug_EraseResult == W25Q64_OK) &&
        (Debug_ProgramResult == W25Q64_OK) &&
        (Debug_ReadResult == W25Q64_OK) &&
        (Debug_ProgramCheckPassed == 1)) {
        GPIO_ResetBits(GPIOC, GPIO_Pin_13);
    } else {
        GPIO_SetBits(GPIOC, GPIO_Pin_13);
    }

    while (1) {
    }
}
```

**W25Q64.c**
```c
#include "W25Q64.h"
#include "Delay.h"

#define W25Q64_CMD_WRITE_ENABLE 	0x06
#define W25Q64_CMD_WRITE_DISABLE	0x04
#define W25Q64_CMD_READ_STATUS_REG1 0x05
#define W25Q64_CMD_READ_DATA		0x03
#define W25Q64_CMD_PAGE_PROGRAM     0x02
#define W25Q64_CMD_SECTOR_ERASE     0x20
#define W25Q64_CMD_ENABLE_RESET    0x66
#define W25Q64_CMD_RESET_DEVICE    0x99

/*
 * W25Q64JV 容量：
 *
 * 64 Mbit = 8 MByte = 0x800000 字节
 *
 * 有效地址：
 * 0x000000 ～ 0x7FFFFF
 */
 
 #define W25Q64_CAPACITY_BYTES   0x800000UL
 #define W25Q64_PAGE_SIZE        256U
 #define W25Q64_SECTOR_SIZE      4096UL
 #define W25Q64_BUSY_TIMEOUT     0x800000UL

W25Q64_Result W25Q64_SwapByte(uint8_t SendData, uint8_t* ReceiveData) {
	
	if(MySPI_SwapByte(SendData, ReceiveData) != MySPI_OK) {
		return W25Q64_ERROR_SPI;
	}
	return W25Q64_OK;
}

static W25Q64_Result W25Q64_SendByte(uint8_t Data) {
	uint8_t Dummy;
	return W25Q64_SwapByte(Data, &Dummy);

}

static W25Q64_Result W25Q64_EndTransaction(W25Q64_Result CurrentResult) {
	if (MySPI_Stop() != MySPI_OK) return W25Q64_ERROR_SPI;
	
	return CurrentResult;
}

//发送24位地址
static W25Q64_Result W25Q64_SendAddress(uint32_t Address) {
	W25Q64_Result Result;
	
	Result = W25Q64_SendByte((uint8_t)(Address >> 16));
	if (Result != W25Q64_OK) return Result;
	
	Result = W25Q64_SendByte((uint8_t)(Address >> 8));
	if (Result != W25Q64_OK) return Result;
	
	Result = W25Q64_SendByte((uint8_t)Address);
	if (Result != W25Q64_OK) return Result;
	
	return Result;
}

//查看地址是否溢出
static uint8_t W25Q64_IsAddressRangeValid(uint32_t Address, uint32_t Length) {
	if (Address >= W25Q64_CAPACITY_BYTES) return 0;
	if (Length > (W25Q64_CAPACITY_BYTES - Address)) return 0;
	
	return 1;
}

//读SR1
W25Q64_Result W25Q64_ReadStatusRegister1(uint8_t* Status){
	W25Q64_Result Result;
	MySPI_Start();
	
	Result = W25Q64_SendByte(W25Q64_CMD_READ_STATUS_REG1);
	if (Result == W25Q64_OK) {
		Result = W25Q64_SwapByte(0xff, Status);
	}
	
	return W25Q64_EndTransaction(Result);
}


W25Q64_Result W25Q64_WriteEnable(void){
	W25Q64_Result Result;
	uint8_t Status;
	MySPI_Start();
	
	Result = W25Q64_SendByte(W25Q64_CMD_WRITE_ENABLE);
	Result = W25Q64_EndTransaction(Result);
	if (Result != W25Q64_OK) return Result;
	
	Result = W25Q64_ReadStatusRegister1(&Status);
	if (Result != W25Q64_OK) return Result;
	
	if ((Status & 0x02) == 0) return W25Q64_ERROR_WRITE_ENABLE;
	
	return W25Q64_OK;
}

W25Q64_Result W25Q64_WriteDisable(void){
	W25Q64_Result Result;
	MySPI_Start();
	
	Result = W25Q64_SendByte(W25Q64_CMD_WRITE_DISABLE);

	return W25Q64_EndTransaction(Result);
}

//查看是否处于擦写状态
W25Q64_Result W25Q64_WaitBusy(void) {
	W25Q64_Result Result;
	uint8_t Status;
	uint32_t Timeout;
	
	Timeout = W25Q64_BUSY_TIMEOUT;
	while (Timeout > 0) {
		Result = W25Q64_ReadStatusRegister1(&Status);
		if (Result != W25Q64_OK) return Result;
		if ((Status & 0x01) == 0) return W25Q64_OK;
		--Timeout;
	}
	return W25Q64_ERROR_BUSY_TIMEOUT;
}

//擦除扇区
W25Q64_Result W25Q64_SectorErase(uint32_t Address) {
	W25Q64_Result Result;
	//判断地址是否合法
	if (Address >= W25Q64_CAPACITY_BYTES) return W25Q64_ERROR_ADDRESS;
	if ((Address & 0xFFFUL) != 0) return W25Q64_ERROR_SECTOR_ALIGMENT;
	//写使能
	Result = W25Q64_WriteEnable();
	if (Result != W25Q64_OK) return Result;
	MySPI_Start();
	//发送指令擦除扇区
	Result = W25Q64_SendByte(W25Q64_CMD_SECTOR_ERASE);
	if (Result != W25Q64_OK) return Result;
	Result = W25Q64_SendAddress(Address);
	if (Result != W25Q64_OK) return Result;
	//结束
	Result = W25Q64_EndTransaction(Result);
	if (Result != W25Q64_OK) return Result;
	return W25Q64_WaitBusy();
}

//页编程
W25Q64_Result W25Q64_PageProgram(uint32_t Address, const uint8_t* Data, uint16_t Length) {
	W25Q64_Result Result;
	//判断参数是否合法
	if (Length == 0 || Length > W25Q64_PAGE_SIZE) return W25Q64_ERROR_LENGTH;
	if (W25Q64_IsAddressRangeValid(Address, Length) != 1) return W25Q64_ERROR_ADDRESS;

	uint32_t OffSet = (uint16_t)(Address & 0xff);  //获取第一页起始地址
	if (OffSet + Length> W25Q64_PAGE_SIZE) return W25Q64_ERROR_PAGE_BOUNDARY;  //检查该页写入数据后是否超过了页大小
	//开始页编程
	Result = W25Q64_WriteEnable();
	if (Result != W25Q64_OK) return Result;
	MySPI_Start();
	Result = W25Q64_SendByte(W25Q64_CMD_PAGE_PROGRAM);
	if (Result != W25Q64_OK) return Result;
	Result = W25Q64_SendAddress(Address);  //页编程起点
	if (Result != W25Q64_OK) return Result;
	//开始写入数据
	for (uint16_t i = 0; i < Length && Result == W25Q64_OK; ++i ) {
		Result = W25Q64_SendByte(Data[i]);
	}
	Result = W25Q64_EndTransaction(Result);
	if (Result != W25Q64_OK) return Result;
	return W25Q64_WaitBusy();
}

//跨页编程
W25Q64_Result W25Q64_WriteDataNoErase(uint32_t Address, const uint8_t* Data, uint32_t Length){
	uint16_t WriteLength;
	W25Q64_Result Result;
	
	while (Length > 0) {
		uint16_t Offset = Address & 0xff;
		uint32_t PageRemaining = W25Q64_PAGE_SIZE - Offset;
		if (Length > PageRemaining) {
			WriteLength = PageRemaining;
		}else {
			WriteLength = Length;
		}
		Result = W25Q64_PageProgram(Address, Data, WriteLength);
		
		if (Result != W25Q64_OK) return Result;
		
		Data += WriteLength;
		Length -= WriteLength;
		Address += WriteLength;
	}
	return W25Q64_OK;
}

W25Q64_Result W25Q64_ReadData(uint32_t Address, uint8_t* Data, uint16_t Length) {
	W25Q64_Result Result;
	//参数检查
	if (Length == 0) return W25Q64_ERROR_LENGTH;
	if (W25Q64_IsAddressRangeValid(Address, Length) != 1) return W25Q64_ERROR_ADDRESS;
	//等待开始 
	Result = W25Q64_WaitBusy();
	if (Result != W25Q64_OK) return Result;
	MySPI_Start();
	//开始读取
	Result = W25Q64_SendByte(W25Q64_CMD_READ_DATA);
	if (Result != W25Q64_OK) return Result;
	Result = W25Q64_SendAddress(Address);
	if (Result != W25Q64_OK) return Result;
	//读取数据
	for (uint16_t i = 0; i < Length && Result == W25Q64_OK; ++i ) {
		Result = W25Q64_SwapByte(0xff, &Data[i]);
	}
	Result = W25Q64_EndTransaction(Result);
	if (Result != W25Q64_OK) return Result;
	return W25Q64_WaitBusy();
}

//增加软件复位
W25Q64_Result W25Q64_Reset(void){
    W25Q64_Result Result;

    /*
     * 第一步：
     * Enable Reset = 0x66
     */
    MySPI_Start();

    Result = W25Q64_SendByte(
        W25Q64_CMD_ENABLE_RESET
    );

    Result = W25Q64_EndTransaction(
        Result
    );

    if (Result != W25Q64_OK){
        return Result;
    }

    /*
     * 第二步：
     * Reset Device = 0x99
     *
     * 66 和 99 中间不能插入其他 W25Q64 指令。
     */
    MySPI_Start();

    Result = W25Q64_SendByte(
        W25Q64_CMD_RESET_DEVICE
    );

    Result = W25Q64_EndTransaction(
        Result
    );

    if (Result != W25Q64_OK){
        return Result;
    }

    /*
     * 数据手册 tRST 约 30 us。
     *
     * 当前没有必要专门做 30 us 延时，
     * 直接等 1 ms 足够用于实验。
     */
    Delay_ms(1);

    return W25Q64_OK;
}
```
