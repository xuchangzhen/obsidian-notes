本实验读取了W25Q64的寄存器1状态，并打开写入锁(发送0x06)

**main.c**（AI）
```c
#include "stm32f10x.h"
#include "MySPI.h"
#include "W25Q64.h"

/*
 * 三次读取到的完整状态寄存器值。
 */
volatile uint8_t Debug_StatusBefore;
volatile uint8_t Debug_StatusAfterEnable;
volatile uint8_t Debug_StatusAfterDisable;

/*
 * 从状态寄存器中单独提取出的状态位。
 */
volatile uint8_t Debug_BUSYBefore;
volatile uint8_t Debug_WELBefore;

volatile uint8_t Debug_BUSYAfterEnable;
volatile uint8_t Debug_WELAfterEnable;

volatile uint8_t Debug_BUSYAfterDisable;
volatile uint8_t Debug_WELAfterDisable;

/*
 * 函数执行结果。
 */
volatile W25Q64_Result Debug_ResultRead1;
volatile W25Q64_Result Debug_ResultEnable;
volatile W25Q64_Result Debug_ResultRead2;
volatile W25Q64_Result Debug_ResultDisable;
volatile W25Q64_Result Debug_ResultRead3;


static void LED_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStructure;

    RCC_APB2PeriphClockCmd(
        RCC_APB2Periph_GPIOC,
        ENABLE
    );

    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_13;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_PP;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_2MHz;

    GPIO_Init(GPIOC, &GPIO_InitStructure);

    /*
     * PC13 高电平：熄灭。
     */
    GPIO_SetBits(GPIOC, GPIO_Pin_13);
}


int main(void)
{
    uint8_t StatusBefore;
    uint8_t StatusAfterEnable;
    uint8_t StatusAfterDisable;

    LED_Init();
    MySPI_Init();

    /*
     * 第一次读取：
     * 查看初始状态。
     */
    Debug_ResultRead1 =
        W25Q64_ReadStatusRegister1(
            &StatusBefore
        );

    /*
     * 发送 Write Enable：0x06。
     */
    Debug_ResultEnable =
        W25Q64_WriteEnable();

    /*
     * 第二次读取：
     * 验证 WEL 是否变为 1。
     */
    Debug_ResultRead2 =
        W25Q64_ReadStatusRegister1(
            &StatusAfterEnable
        );

    /*
     * 发送 Write Disable：0x04。
     */
    Debug_ResultDisable =
        W25Q64_WriteDisable();

    /*
     * 第三次读取：
     * 验证 WEL 是否重新变为 0。
     */
    Debug_ResultRead3 =
        W25Q64_ReadStatusRegister1(
            &StatusAfterDisable
        );

    /*
     * 保存完整状态字节。
     */
    Debug_StatusBefore =
        StatusBefore;

    Debug_StatusAfterEnable =
        StatusAfterEnable;

    Debug_StatusAfterDisable =
        StatusAfterDisable;

    /*
     * bit0：BUSY
     */
    Debug_BUSYBefore =
        StatusBefore & 0x01;

    Debug_BUSYAfterEnable =
        StatusAfterEnable & 0x01;

    Debug_BUSYAfterDisable =
        StatusAfterDisable & 0x01;

    /*
     * bit1：WEL
     *
     * 先与 0x02 按位与，
     * 再右移 1 位，得到 0 或 1。
     */
    Debug_WELBefore =
        (StatusBefore & 0x02) >> 1;

    Debug_WELAfterEnable =
        (StatusAfterEnable & 0x02) >> 1;

    Debug_WELAfterDisable =
        (StatusAfterDisable & 0x02) >> 1;

    /*
     * 判断实验结果。
     */
    if ((Debug_ResultRead1 == W25Q64_OK) &&
        (Debug_ResultEnable == W25Q64_OK) &&
        (Debug_ResultRead2 == W25Q64_OK) &&
        (Debug_ResultDisable == W25Q64_OK) &&
        (Debug_ResultRead3 == W25Q64_OK) &&

        (Debug_BUSYBefore == 0) &&
        (Debug_BUSYAfterEnable == 0) &&
        (Debug_BUSYAfterDisable == 0) &&

        (Debug_WELBefore == 0) &&
        (Debug_WELAfterEnable == 1) &&
        (Debug_WELAfterDisable == 0))
    {
        /*
         * 全部正确：点亮 PC13。
         */
        GPIO_ResetBits(
            GPIOC,
            GPIO_Pin_13
        );
    }
    else
    {
        GPIO_SetBits(
            GPIOC,
            GPIO_Pin_13
        );
    }

    while (1)
    {
    }
}
```

**W25Q64.c**
```c
#include "W25Q64.h"

#define W25Q64_CMD_WRITE_ENABLE 	0x06
#define W25Q64_CMD_WRITE_DISABLE	0x04
#define W25Q64_CMD_READ_STATUS_REG1 0x05

W25Q64_Result W25Q64_SwapByte(uint8_t SendData, uint8_t* ReceiveData) {
	
	if(MySPI_SwapByte(SendData, ReceiveData) != MySPI_OK) {
		return W25Q64_ERROR_SPI;
	}
	return W25Q64_OK;
}

W25Q64_Result W25Q64_ReadStatusRegister1(uint8_t* Status){
	uint8_t Dummy = 0xff;
	MySPI_Start();
	
	if (W25Q64_SwapByte(W25Q64_CMD_READ_STATUS_REG1, &Dummy) != W25Q64_OK) {
		MySPI_Stop();
		return W25Q64_ERROR_SPI;
	}
	
	if (W25Q64_SwapByte(0xff, Status) != W25Q64_OK) {
		MySPI_Stop();
		return W25Q64_ERROR_SPI;
	}
	
	if (MySPI_Stop() != MySPI_OK) return W25Q64_ERROR_SPI;
	
	return W25Q64_OK;
}

W25Q64_Result W25Q64_WriteEnable(void){
	uint8_t Dummy;
	MySPI_Start();
	
	if (W25Q64_SwapByte(W25Q64_CMD_WRITE_ENABLE, &Dummy) != W25Q64_OK) {
		MySPI_Stop();
		return W25Q64_ERROR_SPI;
	}
	if (MySPI_Stop() != MySPI_OK) return W25Q64_ERROR_SPI;
	
	return W25Q64_OK;
}

W25Q64_Result W25Q64_WriteDisable(void){
	uint8_t Dummy;
	MySPI_Start();
	
	if (W25Q64_SwapByte(W25Q64_CMD_WRITE_DISABLE, &Dummy) != W25Q64_OK) {
		MySPI_Stop();
		return W25Q64_ERROR_SPI;
	}
	if (MySPI_Stop() != MySPI_OK) return W25Q64_ERROR_SPI;
	
	return W25Q64_OK;
}

```

这里我们可以了解一下SPI的工作模式
虽然是全双工，但是发送一个8bit的指令，再等待其返回一个8bit的结果，需要16个时钟周期
因为需要花费8个时钟周期发送数据，待slave收到指令之后，返回结果也需要花费8个时钟周期
我们看到在Swap里面，我们接收用了一些占位字符Dummy或者0xff
因为有时候只是发送了数据，但是并没有接收的数据，就需要一个占位符
有时候接收数据，但是并没有数据需要发送，就需要发送一个Dummy数据占位，通常设置为0xff，因为时钟信号由Master产生，所以需要假装发送一个数据，实际上这个数据并没有实际意义，只是为了产生时钟信号
