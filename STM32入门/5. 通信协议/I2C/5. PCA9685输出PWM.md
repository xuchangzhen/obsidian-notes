**PCA9685.h**
```c
#ifndef __PCA9685_H
#define __PCA9685_H

#include "stm32f10x.h"                  // Device header

#define PCA9685_ADDRESS   	    0x40
#define PCA9685_MODE1     	    0x00
#define PCA9685_MODE2	   		0x01
#define PCA9685_LED0_ON_L		0x06
#define PCA9685_LED0_ON_H		0x07
#define PCA9685_LED0_OFF_L		0x08
#define PCA9685_LED0_OFF_H		0x09
#define PCA9685_PRE_SCALE		0xFE
#define PCA9685_PRESCALE_50HZ	121

typedef enum{
    PCA9685_OK = 0,

    PCA9685_ERROR_PARAMETER,

    PCA9685_ERROR_MODE1_INIT,
    PCA9685_ERROR_MODE2_INIT,

    PCA9685_ERROR_MODE1_READ,
    PCA9685_ERROR_ENTER_SLEEP,
    PCA9685_ERROR_PRE_SCALE,
    PCA9685_ERROR_EXIT_SLEEP,

    PCA9685_ERROR_FULL_OFF,
    PCA9685_ERROR_ON_L,
    PCA9685_ERROR_ON_H,
    PCA9685_ERROR_OFF_L,
    PCA9685_ERROR_OFF_H

}PCA9685_Result;

PCA9685_Result PCA9685_Init(void);
PCA9685_Result PCA9685_SetFrequency50Hz(void);
PCA9685_Result PCA9685_SetPWM(uint8_t Channel, uint16_t OnCount, uint16_t OffCount);

#endif
```

**PCA9685.c**
```c
#include "PCA9685.h"
#include "MyI2C.h"
#include "Delay.h"

PCA9685_Result PCA9685_SetFrequency50Hz(void) {  //设置pwm频率为50Hz(20ms) 
	//读取mode1 -> 修改sleep位进入睡眠 -> 睡眠写入mode1 -> 修改PRESCALE -> 关闭sleep并等待内部振荡器唤醒
	uint8_t OldMode;
	uint8_t SleepMode;
	uint8_t WakeMode;
	
	if (MyI2C_ReadRegister(PCA9685_ADDRESS, PCA9685_MODE1, &OldMode) != MyI2C_OK) return PCA9685_ERROR_MODE1_READ;
	SleepMode = OldMode | 0x10;
	if (MyI2C_WriteRegister(PCA9685_ADDRESS, PCA9685_MODE1, SleepMode) != MyI2C_OK) return PCA9685_ERROR_ENTER_SLEEP;
	if (MyI2C_WriteRegister(PCA9685_ADDRESS, PCA9685_PRE_SCALE, PCA9685_PRESCALE_50HZ) != MyI2C_OK) return PCA9685_ERROR_PRE_SCALE;
	WakeMode = OldMode & 0xEF;
	if (MyI2C_WriteRegister(PCA9685_ADDRESS, PCA9685_MODE1, WakeMode) != MyI2C_OK) return PCA9685_ERROR_EXIT_SLEEP;
	
	Delay_ms(1);
 	return PCA9685_OK;
}

//开启AI并sleep = 0 -> mode2 INVRT = 0, OCH = 0, OUTDRV = 1;
PCA9685_Result PCA9685_Init(void) {
	if (MyI2C_WriteRegister(PCA9685_ADDRESS, PCA9685_MODE1, 0x20) != MyI2C_OK) return PCA9685_ERROR_MODE1_INIT;
	if (MyI2C_WriteRegister(PCA9685_ADDRESS, PCA9685_MODE2, 0x04) != MyI2C_OK) return PCA9685_ERROR_MODE2_INIT;
	return PCA9685_SetFrequency50Hz();
}

//判断参数是否合理 -> 先开启full off后写入起始和终点
PCA9685_Result PCA9685_SetPWM(uint8_t Channel, uint16_t OnCount, uint16_t OffCount) {
	uint8_t RegisterAddress;
	uint8_t OnLow;
	uint8_t OnHigh;
	uint8_t OffLow;
	uint8_t OffHigh;
	
	if (Channel > 15 || OnCount > 4095 || OffCount > 4095) return PCA9685_ERROR_PARAMETER;

	RegisterAddress = PCA9685_LED0_ON_L + Channel * 4;
	OnLow = OnCount & 0xff;
	OnHigh = (OnCount >> 8) & 0x0f;
	OffLow = OffCount & 0xff;
	OffHigh = (OffCount >> 8) & 0x0f;
	
	if (MyI2C_WriteRegister(PCA9685_ADDRESS, RegisterAddress + 0, OnLow) != MyI2C_OK) return PCA9685_ERROR_ON_L;
	if (MyI2C_WriteRegister(PCA9685_ADDRESS, RegisterAddress + 1, OnHigh) != MyI2C_OK) return PCA9685_ERROR_ON_H;
	if (MyI2C_WriteRegister(PCA9685_ADDRESS, RegisterAddress + 2, OffLow) != MyI2C_OK) return PCA9685_ERROR_OFF_L;
	if (MyI2C_WriteRegister(PCA9685_ADDRESS, RegisterAddress + 3, OffHigh) != MyI2C_OK) return PCA9685_ERROR_ON_H;
	
	return PCA9685_OK;
}


```

PCA9685的计数器是12位，可以计数0~4095，也就是4096个数
通过LED_ON和LED_OFF可以控制计数的起点和终点
ON为计数的开始，OFF为计数的结束
LED_ON_LOW代表计数的低八位，LED_ON_HIGH代表计数的高四位

咦？这不对吧，怎么周期偏少那么多
这是因为内部振荡器频率是会有误差的，可能偏高可能偏低
这里我们就可以知道我们的内部振荡器频率是偏高了

通过公式反推振荡器频率
f_osc = f_pwm × 4096 × (PRE_SCALE + 1)
其中f_pwm = 1 / T_PWM =  1 / 17.659274  * 1000；==注意单位==
大概其中f_pwm = 56.6Hz
f_osc ≈ 56.63 × 4096 × 122
      ≈ 28.30 MHz

重新计算PRESCALE
PRE_SCALE
= round(28,300,000 / (4096 × 50)) - 1
≈ 137
