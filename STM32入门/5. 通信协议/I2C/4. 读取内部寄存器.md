```c
	GPIO_InitStrucure.GPIO_Speed = GPIO_Speed_50MHz;
```

**MyI2C.c**
```c
#include "stm32f10x.h"                  // Device header
#include "MyI2C.h"

#define MyI2C_TIMEOUT 100000UL


void MyI2C_Init(void) {
	
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOB | RCC_APB2Periph_AFIO, ENABLE);
	RCC_APB1PeriphClockCmd(RCC_APB1Periph_I2C1, ENABLE);
	
	GPIO_InitTypeDef GPIO_InitStructure;
	I2C_InitTypeDef I2C_InitStructure;
	
	I2C_DeInit(I2C1);
	
	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_OD;
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_6 | GPIO_Pin_7;
	GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
	GPIO_Init(GPIOB, &GPIO_InitStructure);
	
	I2C_InitStructure.I2C_Mode = I2C_Mode_I2C;
	I2C_InitStructure.I2C_DutyCycle = I2C_DutyCycle_2;
	I2C_InitStructure.I2C_OwnAddress1 = 0x00;
	I2C_InitStructure.I2C_Ack = I2C_Ack_Enable;
	I2C_InitStructure.I2C_AcknowledgedAddress = I2C_AcknowledgedAddress_7bit;
	I2C_InitStructure.I2C_ClockSpeed = 100000;
	I2C_Init(I2C1, &I2C_InitStructure);
	
	I2C_Cmd(I2C1, ENABLE);	
}

MyI2C_Status MyI2C_CheckAddress(uint8_t Address7Bit) {
	uint32_t Timeout;
	Timeout = MyI2C_TIMEOUT;
	
	//总线忙 无法连接
	while (I2C_GetFlagStatus(I2C1, I2C_FLAG_BUSY) == SET) {
		if (--Timeout == 0) return MyI2C_ERROR_BUSY;
	}
	
	I2C_GenerateSTART(I2C1, ENABLE);
	
	//等待EV5(Start Success)
	Timeout = MyI2C_TIMEOUT;
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_MODE_SELECT) == ERROR) {
		if (--Timeout == 0) {
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_START;
		}
	}
	
	//开始向地址发送
	Timeout = MyI2C_TIMEOUT;
	I2C_Send7bitAddress(
		I2C1,
		Address7Bit << 1,
		I2C_Direction_Transmitter
	);
	
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED) ==  ERROR) {
		
		if (I2C_GetFlagStatus(I2C1, I2C_FLAG_AF) ==  SET) {  //从机没有拉低SDA，表示未应答
			I2C_ClearFlag(I2C1, I2C_FLAG_AF);
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_NACK;
		}
		
		if (--Timeout == 0) {  //从机拉低了电平，但是时间太久，超时了
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_ADDRESS_TIMEOUT;
		}
	}
	I2C_GenerateSTOP(I2C1, ENABLE);
	return MyI2C_OK;
}

static void MyI2C_StopAndClearFlagAF(void) {
	I2C_GenerateSTOP(I2C1, ENABLE);
	
	if (I2C_GetFlagStatus(I2C1, I2C_FLAG_AF) == SET) {
		I2C_ClearFlag(I2C1, I2C_FLAG_AF);
	}
}

//先写入数据
MyI2C_Status MyI2C_WriteRegister(uint8_t Address7Bit, uint8_t RegisterAddress, uint8_t Data ) {
	uint32_t Timeout;
	Timeout = MyI2C_TIMEOUT;
	
	//总线忙 无法连接
	while (I2C_GetFlagStatus(I2C1, I2C_FLAG_BUSY) == SET) {
		if (--Timeout == 0) return MyI2C_ERROR_BUSY;
	}
	
	I2C_GenerateSTART(I2C1, ENABLE);
	
	//等待EV5(Start Success)
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_MODE_SELECT) == ERROR) {
		if (--Timeout == 0) {
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_START;
		}
	}
	
	//开始向地址发送
	Timeout = MyI2C_TIMEOUT;
	I2C_Send7bitAddress(
		I2C1,
		Address7Bit << 1,
		I2C_Direction_Transmitter
	);
	
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED) ==  ERROR) {
		
		if (I2C_GetFlagStatus(I2C1, I2C_FLAG_AF) ==  SET) {  //从机没有拉低SDA，表示未应答
			I2C_ClearFlag(I2C1, I2C_FLAG_AF);
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_NACK;
		}
		
		if (--Timeout == 0) {  //从机拉低了电平，但是时间太久，超时了
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_ADDRESS_TIMEOUT;
		}
	}
	
	//发送从机的内部寄存器地址
	I2C_SendData(I2C1, RegisterAddress);
	Timeout = MyI2C_TIMEOUT;
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_BYTE_TRANSMITTING) == ERROR) {
		if (I2C_GetFlagStatus(I2C1, I2C_FLAG_AF) ==  SET) {
			MyI2C_StopAndClearFlagAF();
			return MyI2C_ERROR_REGISTER_NACK;
		}
		
		if (--Timeout == 0) {
			MyI2C_StopAndClearFlagAF();
			return MyI2C_ERROR_REGISTER_TIMEOUT;
		}
	}
	
	//向内部地址发送数据
	I2C_SendData(I2C1, Data);
	Timeout = MyI2C_TIMEOUT;
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_BYTE_TRANSMITTED) == ERROR) {
		if (I2C_GetFlagStatus(I2C1, I2C_FLAG_AF) == SET) {
			MyI2C_StopAndClearFlagAF();
			return MyI2C_ERROR_DATA_NACK;
		}
		
		if (--Timeout == 0) {
			MyI2C_StopAndClearFlagAF();
			return MyI2C_ERROR_DATA_TIMEOUT;
		}
	}
	I2C_GenerateSTOP(I2C1, ENABLE);
	return MyI2C_OK;
}

MyI2C_Status MyI2C_ReadRegister(uint8_t Address7Bit, uint8_t RegisterAddress, uint8_t* Data) {
	uint32_t Timeout = MyI2C_TIMEOUT;
	volatile uint16_t Temp;
	
	//避免传入空指针
	if (Data == 0) return MyI2C_ERROR_RECEIVE_TIMEOUT;
	
	//开始前恢复ack，因为单字节接收会暂时关闭ACK，但是有必要吗？ 
	//答: 是需要的，软件规定了有ACK应答位，但是并不代表硬件会自动应答，所以要开启
	I2C_AcknowledgeConfig(I2C1, ENABLE);
	
	//等待总线空闲
	while (I2C_GetFlagStatus(I2C1, I2C_FLAG_BUSY) == SET) {
		if (--Timeout == 0) return MyI2C_ERROR_BUSY;
	}
	
	//清除上次的AF,上次是什么时候？ 
	//答：因为函数错误的时候可能没有清除错误标志，下次进来可能会带着错误的标志进来，但是其实不需要也行，因为我们在错误退出的位置都清除了AF，所以这里只能起到防御的作用了
	if (I2C_GetFlagStatus(I2C1, I2C_FLAG_AF) == SET) {
		MyI2C_StopAndClearFlagAF();
	}
	
	//开始start 
	I2C_GenerateSTART(I2C1, ENABLE);
	Timeout = MyI2C_TIMEOUT;
	
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_MODE_SELECT) == ERROR) {  //未能进入EV5
		if (--Timeout == 0) {
			I2C_GenerateSTOP(I2C1, ENABLE);  
			return MyI2C_ERROR_START;
		}
	}
	
	//发送接收端地址 进入EV6
	Timeout = MyI2C_TIMEOUT;
	I2C_Send7bitAddress(I2C1, Address7Bit << 1, I2C_Direction_Transmitter);
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED) == ERROR) {
		if (I2C_GetFlagStatus(I2C1, I2C_FLAG_AF) == SET) {
			MyI2C_StopAndClearFlagAF();
			return MyI2C_ERROR_NACK;
		}
		if (--Timeout == 0) {
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_ADDRESS_TIMEOUT;
		}
	}
	
	//发送内部寄存器地址并等待EV8_2
	I2C_SendData(I2C1, RegisterAddress);
	Timeout = MyI2C_TIMEOUT;
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_BYTE_TRANSMITTED) == ERROR) {
		if (I2C_GetFlagStatus(I2C1, I2C_FLAG_AF) == SET) {
			MyI2C_StopAndClearFlagAF();
			return MyI2C_ERROR_REGISTER_NACK;
		}
		if (--Timeout == 0) {
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_REGISTER_TIMEOUT;
		}
	}
	
	//接下来开始读取数据
	//先restart
	I2C_GenerateSTART(I2C1, ENABLE);
	Timeout = MyI2C_TIMEOUT;
	
	//看restart是否成功
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_MODE_SELECT) == ERROR) {
		if(--Timeout == 0) {
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_RESTART;
		}
	}
	
	//再次发送设备地址，方向为读
	I2C_Send7bitAddress(I2C1, Address7Bit << 1, I2C_Direction_Receiver);
	Timeout = MyI2C_TIMEOUT;
	//等待 ADDR 置位,这里不能使用I2C_EVENT_MASTER_RECEIVER_MODE_SELECTED，因为会清除ADDR，违背了单字节读取的顺序
	while (I2C_GetFlagStatus(I2C1, I2C_FLAG_ADDR) == RESET) {
		if (I2C_GetFlagStatus(I2C1, I2C_FLAG_AF) == SET) {
			MyI2C_StopAndClearFlagAF();
			return MyI2C_ERROR_READ_ADDRESS_NACK;
		}
		if (--Timeout == 0) {
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_READ_ADDRESS_TIMEOUT;
		}
	}
	
	I2C_AcknowledgeConfig(I2C1, DISABLE);  //关闭ACK
	
	//清除ADDR
	Temp = I2C1->SR1;
	Temp = I2C1->SR2;
	(void)Temp;  //告诉编译器这个值不会继续使用
	I2C_GenerateSTOP(I2C1, ENABLE);
	
	Timeout = MyI2C_TIMEOUT;
	while (I2C_GetFlagStatus(I2C1, I2C_FLAG_RXNE) == RESET) {  //等待数据到达RX寄存器
		if (--Timeout == 0) {
			I2C_AcknowledgeConfig(I2C1, ENABLE);  //为下次通信打开ack
			return MyI2C_ERROR_RECEIVE_TIMEOUT; 
		}
	}
	
	*Data = I2C_ReceiveData(I2C1);  
	
	//等待STOP成功
	Timeout = MyI2C_TIMEOUT;
	while (I2C_GetFlagStatus(I2C1, I2C_FLAG_BUSY) == SET) {
		if (--Timeout == 0) return MyI2C_ERROR_STOP_TIMEOUT;
	}
	I2C_AcknowledgeConfig(I2C1, ENABLE);
	
	return MyI2C_OK;
	
}
```

**读取数据后关闭流程**

等待 ADDR = 1
↓
关闭 ACK
↓
读 SR1、读 SR2，清除 ADDR
↓
设置 STOP
↓
等待 RXNE = 1
↓
读取 DR
↓
等待 STOP 完成
↓
恢复 ACK
> STOP 必须在读取数据之前设置，但必须在关闭 ACK、清除 ADDR之后设置。

**为什么在MyI2C_ReadRegister还要有一次写方向的通信？**
因为我们需要选择内部寄存器，如果只是
```c
I2C_Send7bitAddress(I2C1, Address7Bit << 1, I2C_Direction_Receiver);
```
此时我们并不知道选择的是PCA9685的哪一个内部寄存器

**为什么要Restart？**
I2C通信中，第一次写入的数据代表选择哪个通信设备，第二次写入的数据为选择内部寄存器，接下来的操作才是写值
如果在选择了内部寄存器之后，没有restart直接执行
```c
I2C_Send7bitAddress(
    I2C1,
    Address7Bit << 1,
    I2C_Direction_Transmitter
);
```
PCA9685的接收寄存器就会认为你需要写入数据0x80
那不就完蛋了吗

然后我们因为没有STOP释放总线，所以通信没有结束，此时内部寄存器指针还是指向着之前第二次写入数据选择的内部寄存器，所以ReStart也没有释放总线，可以直接读取到内部寄存器指针指向的地址的数据
