**MyI2C.c**
```c
#include "stm32f10x.h"                  // Device header
#include "MyI2C.h"

#define MyI2C_TIMEOUT 100000UL


void MyI2C_Init(void) {
	
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOB | RCC_APB2Periph_AFIO, ENABLE);
	RCC_APB1PeriphClockCmd(RCC_APB1Periph_I2C1, ENABLE);
	
	GPIO_InitTypeDef GPIO_InitStructure;
	I2C_InitTypeDef I2C_InitStructure;
	
	I2C_DeInit(I2C1);
	
	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_OD;
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_6 | GPIO_Pin_7;
	GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
	GPIO_Init(GPIOB, &GPIO_InitStructure);
	
	I2C_InitStructure.I2C_Mode = I2C_Mode_I2C;
	I2C_InitStructure.I2C_DutyCycle = I2C_DutyCycle_2;
	I2C_InitStructure.I2C_OwnAddress1 = 0x00;
	I2C_InitStructure.I2C_Ack = I2C_Ack_Enable;
	I2C_InitStructure.I2C_AcknowledgedAddress = I2C_AcknowledgedAddress_7bit;
	I2C_InitStructure.I2C_ClockSpeed = 100000;
	I2C_Init(I2C1, &I2C_InitStructure);
	
	I2C_Cmd(I2C1, ENABLE);	
}

MyI2C_Status MyI2C_CheckAddress(uint8_t Address7Bit) {
	uint32_t Timeout;
	Timeout = MyI2C_TIMEOUT;
	
	//总线忙 无法连接
	while (I2C_GetFlagStatus(I2C1, I2C_FLAG_BUSY) == SET) {
		if (--Timeout == 0) return MyI2C_ERROR_BUSY;
	}
	
	I2C_GenerateSTART(I2C1, ENABLE);
	
	//等待EV5(Start Success)
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_MODE_SELECT) == ERROR) {
		if (--Timeout == 0) {
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_START;
		}
	}
	
	//开始向地址发送
	Timeout = MyI2C_TIMEOUT;
	I2C_Send7bitAddress(
		I2C1,
		Address7Bit << 1,
		I2C_Direction_Transmitter
	);
	
	while (I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED) ==  ERROR) {
		
		if (I2C_GetFlagStatus(I2C1, I2C_FLAG_AF) ==  SET) {  //从机没有拉低SDA，表示未应答
			I2C_ClearFlag(I2C1, I2C_FLAG_AF);
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_NACK;
		}
		
		if (--Timeout == 0) {  //从机拉低了电平，但是时间太久，超时了
			I2C_GenerateSTOP(I2C1, ENABLE);
			return MyI2C_ERROR_ADDRESS_TIMEOUT;
		}
	}
	I2C_GenerateSTOP(I2C1, ENABLE);
	return MyI2C_OK;
}
```
