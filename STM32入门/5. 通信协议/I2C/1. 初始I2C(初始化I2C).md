本次操作共有三个文件

**main.c**
```c
#include "stm32f10x.h"                  // Device header
#include "MyI2C.h"

volatile uint8_t I2C_SCL_Level;
volatile uint8_t I2C_SDL_Level;

void LED_Init(void) {
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC, ENABLE);
	
	GPIO_InitTypeDef GPIO_InitStrucure;
	GPIO_InitStrucure.GPIO_Mode = GPIO_Mode_Out_PP;
	GPIO_InitStrucure.GPIO_Pin = GPIO_Pin_13;
	GPIO_InitStrucure.GPIO_Speed = GPIO_Speed_50MHz;
	GPIO_Init(GPIOC, &GPIO_InitStrucure);
	
	GPIO_SetBits(GPIOC, GPIO_Pin_13);
}

int main(void) {

	LED_Init();
	MyI2C_Init();
	
	
	while (1) {
		I2C_SCL_Level = GPIO_ReadInputDataBit(GPIOB, GPIO_Pin_6);  //读取GPIO的数据数据寄存器IDR，也就是检测电平状态
		I2C_SDL_Level = GPIO_ReadInputDataBit(GPIOB, GPIO_Pin_7);
		
		if (I2C_SCL_Level && I2C_SDL_Level) {
			GPIO_ResetBits(GPIOC, GPIO_Pin_13);
		}else {
			GPIO_SetBits(GPIOC, GPIO_Pin_13);
		}
	}

}

```

MyI2C.c
```c
#include "stm32f10x.h"                  // Device header
#include "MyI2C.h"

void MyI2C_Init(void) {
	
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOB, ENABLE);
	RCC_APB1PeriphClockCmd(RCC_APB1Periph_I2C1, ENABLE);
	
	GPIO_InitTypeDef GPIO_InitStructure;
	I2C_InitTypeDef I2C_InitStructure;
	
	I2C_DeInit(I2C1);
	
	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_OD;
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_6 | GPIO_Pin_7;
	GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
	GPIO_Init(GPIOA, &GPIO_InitStructure);
	
	I2C_InitStructure.I2C_Mode = I2C_Mode_I2C;
	I2C_InitStructure.I2C_DutyCycle = I2C_DutyCycle_2;
	I2C_InitStructure.I2C_OwnAddress1 = 0x00;
	I2C_InitStructure.I2C_Ack = I2C_Ack_Enable;
	I2C_InitStructure.I2C_AcknowledgedAddress = I2C_AcknowledgedAddress_7bit;
	I2C_InitStructure.I2C_ClockSpeed = 100000;
	I2C_Init(I2C1, &I2C_InitStructure);
	
	I2C_Cmd(I2C1, ENABLE);
	
}


```


本次实验遇到了个问题，就是本来在keil5的User目录下创建了MyI2C.c和MyI2C.h

**I2C_InitStructure**参数介绍
1. I2C_InitStructure.I2C_Mode
在I2C外设下采用哪个总线协议模式
![[Pasted image 20260721021209.png|184]]
	1. I2C_Mode_I2C就是普通的I2C
	普通 I²C 通信中，设备通常分为：

	```
	主机 Master
	从机 Slave
	```
	2.SMBus 全称是：

```
System Management Bus
系统管理总线
```
	SMBus Host 
		↓ 
	发出查询或控制命令 
	SMBus Device 
		↓ 
	返回状态或数据

它是在 I²C 基础上发展出来的一种通信协议，主要用于计算机、电池和电源管理设备。


2. I2C_InitStructure.I2C_DutyCycle
用于设置高低电平的占空比
![[Pasted image 20260721022128.png]]
	1. I2C_DutyCycle_2 ： 低电平 : 高电平为2 ： 1
	2. 16 9就是低电平：高电平 16 ： 9 但是这个参数只在快速模式生效，这个具体看I2C_InitStructure.I2C_ClockSpeed，下面会说到

3. I2C_InitStructure.I2C_OwnAddress1
这个就是用于设置自己作为**从机**时候的地址，注意，不要设置和连接设备的地址一样

4.  I2C_InitStructure.I2C_Ack
设置是否收到数据后自动ACK
![[Pasted image 20260721022612.png]]

5. I2C_InitStructure.I2C_AcknowledgedAddress
设置作为从机的时候自己的地址位数，有两个选项
![[Pasted image 20260721022819.png]]

6. I2C_InitStructure.I2C_ClockSpeed
用于设置SCL的时钟频率
常见速率包括：

```
100 kHz    标准模式
400 kHz    快速模式
1 MHz      快速模式增强版
```
