void LED_Init(void) {
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC, ENABLE);
	
	GPIO_InitTypeDef GPIO_InitStructure;
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_13;
	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_PP;

	GPIO_Init(GPIOC, &GPIO_InitStructure);

	GPIO_SetBits(GPIOC, GPIO_Pin_13);

	GPIO_ResetBits(GPIOC, GPIO_Pin_13);
}

void LED_OFF(void) {
	GPIO_SetBits(GPIOC, GPIO_Pin_13);
}

void USART1_SendChar(char c) {
	while (USART_GetFlagStatus(USART1, USART_FLAG_TXE) == 0);
	USART_SendData(USART1, c);
}

void USART1_SendString(char *str) {
	while (*str) {
		USART1_SendChar(*str);
		str++;
	}
}

void USART1_Init_Config(void) {
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA | RCC_APB2Periph_USART1, ENABLE);
	
	GPIO_InitTypeDef GPIO_InitStructure;
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_9;
	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;
	GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
	GPIO_Init(GPIOA, &GPIO_InitStructure);
	
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_10;
	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IN_FLOATING;
	GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
	GPIO_Init(GPIOA, &GPIO_InitStructure);
	
	USART_InitTypeDef USART_InitStructure;
	USART_InitStructure.USART_BaudRate = 9600;
	USART_InitStructure.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
	USART_InitStructure.USART_Mode = USART_Mode_Tx | USART_Mode_Rx;
	USART_InitStructure.USART_Parity = USART_Parity_No;
	USART_InitStructure.USART_StopBits = USART_StopBits_1;
	USART_InitStructure.USART_WordLength = USART_WordLength_8b;
	USART_Init(USART1, &USART_InitStructure);
	
	USART_ITConfig(USART1, USART_IT_RXNE, ENABLE);
	
	USART_Cmd(USART1, ENABLE);
}

void NVIC_USART1_Init(void) {
	NVIC_PriorityGroupConfig(NVIC_PriorityGroup_2);
	
	NVIC_InitTypeDef NVIC_InitStructure;
	NVIC_InitStructure.NVIC_IRQChannel = USART1_IRQn;
	NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
	NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 1;
	NVIC_InitStructure.NVIC_IRQChannelSubPriority = 1;
	NVIC_Init(&NVIC_InitStructure);
}

int main(void) {
	LED_Init();
	USART1_Init_Config();
	NVIC_USART1_Init();

    USART1_SendString("USART1 interrupt receive test\r\n");
    USART1_SendString("Send 1: LED ON\r\n"); 
    USART1_SendString("Send 0: LED OFF\r\n");
	
	while (1) {
	}
} 

void USART1_IRQHandler(void) {
	if (USART_GetITStatus(USART1, USART_IT_RXNE) != 0 ) {
		char data;
		data = USART_ReceiveData(USART1);
		
		USART1_SendChar(data);
		
        if (data == '1'){
            LED_ON();
            USART1_SendString("\r\nLED ON\r\n");
        }else if (data == '0'){ 
            LED_OFF();
            USART1_SendString("\r\nLED OFF\r\n");
        }
	}
}

这两段代码只有在中断处理函数中多了一行这个代码的区别
```c

```
但是少了这行代码的程序只能进行一次发送，第二次发送的时候无法正确发送，串口助手没有收到回传信息，小灯无变化，那是什么原因导致的呢？

这个原因也是困扰了我一个多月，后面我终于发现了问题所在
我们先来了解一些前置知识

下图是USART->SR(状态地址)对应的每位地址图，下面也有每位的意思

![[Pasted image 20260715205019.png]]

| 位   | 名称     | 英文全称                            | 含义                  |
| --- | ------ | ------------------------------- | ------------------- |
| 0   | `PE`   | Parity Error                    | 奇偶校验错误              |
| 1   | `FE`   | Frame Error                     | 帧格式错误               |
| 2   | `NE`   | Noise Error                     | 噪声错误                |
| 3   | `ORE`  | Overrun Error                   | 接收溢出错误              |
| 4   | `IDLE` | IDLE line detected              | 检测到串口空闲线路           |
| 5   | `RXNE` | Receive Data Register Not Empty | 接收数据寄存器非空           |
| 6   | `TC`   | Transmission Complete           | 整个数据帧发送完成           |
| 7   | `TXE`  | Transmit Data Register Empty    | 发送数据寄存器为空           |
| 8   | `LBD`  | LIN Break Detection Flag        | 检测到 LIN Break 信号    |
| 9   | `CTS`  | Clear To Send Flag              | CTS 状态发生变化,检测引脚电平变化 |

其中我们需要了解的是ORE溢出标志和RXNE接收寄存器
我们在串口助手发送字符 '1' 的时候，其实并不是只发送了字符 '1'
![[Pasted image 20260716001721.png]]
我们打开了发送时加 回车换行
所以我们实际上发的是
```
'1'  '\r'  '\n'
```
三个字符
但是呢，我们一次只能处理一个字符嘛，一个字符一个中断，但是现在三个字符一次性全部涌入，此时ORE就会置1，因为数据寄存器只能存的下一个字符，此时就发生了溢出，然后也是收到了数据，RXNE置1
好那么现在出现自己执行，然后进入if语句当中，当执行到
```c
USART_ReceiveData(USART1)
```
这个时候RXNE ==自动置0==，所以呢我们不需要使用下面这条语句来清除RXNE，虽然正常情况下不会出错，但是我们现在遇到了不正常情况了
```c
USART_ClearITPendingBit(USART1, USART_IT_RXNE);USART_ClearITPendingBit(USART1, USART_IT_RXNE);
```
正是因为下面的这条语句，我们出现了错误
我们来捋一捋，我们实际发送了
```
'1'  '\r'  '\n'
```
当字符'1'被读取之后，本来RXNE也自己置0了，但是后面还有字符进来，这个时候RXNE就会再次置1
但是呢我们执行了
```c
USART_ClearITPendingBit(USART1, USART_IT_RXNE);USART_ClearITPendingBit(USART1, USART_IT_RXNE);
```
就会导致RXNE再次置0，但是ORE还是1
此时就要谈到 ==USART中断条件== 了，满足一点即可
1. RXNE = 1
2. ORE = 1
此时就是满足了ORE = 1
但是呢，此时RXNE为0，进入中断if判断为false，这就尴尬了，啥都没发生，就出来了，但是呢，ORE还是1，又进去，以此反复循环，造成了interrupt storm，也就中断风暴，不断循环进出中断，导致卡死，新的数据也无法进入

了解了那么多，我们可以来看keil的调试watch变量调试信息了

连续发送1， 0 小灯可以正常亮灭 变量值都一致
![[Pasted image 20260715210245.png]]

在加上USART_ClearITPendingBit(USART1, USART_IT_RXNE);之后，发送1亮，0没反应，变量值都不变，rx_data也不变
![[Pasted image 20260715210420.png]]


这里上面两张图我们就可以清晰看到 (这两张图都是先发送1，后发送0，因为发1之后然后再发0两次得到的信息一样，所以就只放了一张) ，第二张图里面，虽然发送了0，但是rx_data还是1，这并不是没收到0，其实已经收到了，只是ORE导致的interrupt storm让0这个数据无法进到中断，导致rx_data的值一直无法更新

### 总结
经过那么多错误，我们也该总结一下了
1. 中断程序尽可能快速，不要在里面执行 send，delay等延时操作，以免等待时间过长或导致溢出
2.
